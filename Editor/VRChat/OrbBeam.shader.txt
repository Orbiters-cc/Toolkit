// N4N0's beam towards its screen (Orbiters contracts, Docs/orb.md, Screen): the veil N4N0 projects its menu with, as the
// speech panel's beam (OrbText's Beam): a faint pyramid from N4N0 to the screen's corners, fine bright lines along its
// edges and a gentle shimmer running from N4N0, thinning out towards the screen. The screen's skinned mesh carries it as
// its second part: the apex rides N4N0 wherever its tether takes it, the base the screen. The second UV is where a vertex
// is across its side and along the beam. Viewers who hide custom shaders see nothing of it (VRCFallback).
// OrbInstaller copies this file as "N4N0 beam.shader".
Shader "Orbiters/N4N0 Beam"
{
    Properties
    {
        _EdgeColor ("Edge", Color) = (0, 0.855, 0.427, 1)
        _Beam ("Beam", Range(0, 1)) = 1
    }
    SubShader
    {
        Tags { "Queue" = "Transparent" "RenderType" = "Transparent" "IgnoreProjector" = "True" "VRCFallback" = "Hidden" }
        Cull Off
        ZWrite Off
        Blend One One

        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile_instancing
            #include "UnityCG.cginc"

            fixed4 _EdgeColor;
            float _Beam;

            struct appdata
            {
                float4 vertex : POSITION;
                float2 beam : TEXCOORD1;
                UNITY_VERTEX_INPUT_INSTANCE_ID
            };

            struct v2f
            {
                float4 pos : SV_POSITION;
                float2 beam : TEXCOORD0;
                UNITY_VERTEX_OUTPUT_STEREO
            };

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_OUTPUT(v2f, o);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                o.pos = UnityObjectToClipPos(v.vertex);
                o.beam = v.beam;
                return o;
            }

            fixed4 frag(v2f i) : SV_Target
            {
                float along = i.beam.y, across = i.beam.x;
                float edges = pow(1 - saturate(min(across, 1 - across) * 2), 28);
                float run = .8 + .2 * sin(along * 14 - _Time.y * 4);
                float fade = smoothstep(.02, .15, along) * (1 - .6 * along);
                return fixed4(_EdgeColor.rgb * (.045 + edges * .5) * run * fade * _Beam, 1);
            }
            ENDCG
        }
    }
}
