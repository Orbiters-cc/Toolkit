// N4N0's pointing rays (Orbiters contracts, Docs/orb.md, Pointing; screen version 7): a thin beam straight out of each
// index fingertip, along the finger, stopping on N4N0's screen with a small dot where it meets it; nothing while the
// finger points elsewhere. The screen's skinned mesh carries it as its third part, bound to the two pointers (one per
// index fingertip): every vertex of a hand's ray sits at its pointer's origin with the pointer's forward as its normal,
// so after skinning each knows the fingertip and the finger's direction in the screen's own space, where the screen is
// the plane z = 0 seen from z < 0. The vertex shader finds where the ray meets that plane and builds the beam (a strip
// facing the camera) and the dot there: no animator, no contact, nothing late. The first UV says where a vertex is on its
// strip or dot, the second whether it is the beam (0) or the dot (1). _Rays (the animator's: on in VR) turns them on.
// Viewers who hide custom shaders see nothing of it (VRCFallback). OrbInstaller copies this file as "N4N0 ray.shader".
Shader "Orbiters/N4N0 Ray"
{
    Properties
    {
        _Color ("Colour", Color) = (0, 0.855, 0.427, 1)
        _Core ("Core", Color) = (1, 1, 1, 1)
        _Rays ("Rays", Range(0, 1)) = 0
        _Width ("Beam width (m)", Float) = 0.0026
        _Dot ("Dot radius (m)", Float) = 0.0055
        _Reach ("Reach (m)", Float) = 2.5
        _Half ("Screen half size", Vector) = (0.24, 0.135, 0, 0)
    }
    SubShader
    {
        Tags { "Queue" = "Transparent+2" "RenderType" = "Transparent" "IgnoreProjector" = "True" "VRCFallback" = "Hidden" }
        Cull Off
        ZWrite Off
        // Premultiplied: the beam adds light (alpha 0), the dot covers what is behind it.
        Blend One OneMinusSrcAlpha

        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile_instancing
            #include "UnityCG.cginc"

            fixed4 _Color, _Core;
            float _Rays, _Width, _Dot, _Reach;
            float4 _Half;

            struct appdata
            {
                float4 vertex : POSITION;
                float3 normal : NORMAL;
                float2 at : TEXCOORD0;
                float2 kind : TEXCOORD1;
                UNITY_VERTEX_INPUT_INSTANCE_ID
            };

            struct v2f
            {
                float4 pos : SV_POSITION;
                // Beam: along (0 at the fingertip, 1 at the screen), across (-1 to 1). Dot: its corner (-1 to 1 each way).
                float2 at : TEXCOORD0;
                float isDot : TEXCOORD1;
                float shown : TEXCOORD2;
                UNITY_VERTEX_OUTPUT_STEREO
            };

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_OUTPUT(v2f, o);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                float3 tip = v.vertex.xyz;
                float3 dir = normalize(v.normal);
                // Metres in the screen's space: the screen may be scaled (its size, the avatar's).
                float scale = length(float3(unity_ObjectToWorld[0].x, unity_ObjectToWorld[1].x, unity_ObjectToWorld[2].x));
                scale = max(scale, 1e-4);
                // In front of the screen, pointing into it, meeting it within reach and over it (a little past its edges).
                float t = dir.z > 0.02 ? -tip.z / dir.z : -1;
                float3 hit = tip + dir * max(t, 0);
                bool shown = _Rays > 0.5 && tip.z < 0 && t > 0 && t * scale < _Reach &&
                    abs(hit.x) < _Half.x + 0.02 / scale && abs(hit.y) < _Half.y + 0.02 / scale;
                o.shown = shown ? 1 : 0;
                o.at = v.at;
                o.isDot = v.kind.x;
                float3 position;
                if (v.kind.x > 0.5)
                {
                    // The dot: on the screen's face, just in front of it.
                    float r = _Dot / scale;
                    position = float3(hit.xy + v.at * r, -0.0015 / scale);
                }
                else
                {
                    // The beam: from the fingertip to just short of the dot's middle, as wide as _Width, facing the camera.
                    float3 end = hit - dir * (_Dot * 0.5 / scale);
                    float3 middle = lerp(tip, end, v.at.x);
                    float3 eye = mul(unity_WorldToObject, float4(_WorldSpaceCameraPos, 1)).xyz;
                    float3 across = cross(end - tip, eye - middle);
                    across = dot(across, across) > 1e-12 ? normalize(across) : float3(1, 0, 0);
                    position = middle + across * v.at.y * (_Width * 0.5 / scale);
                }
                o.pos = shown ? UnityObjectToClipPos(float4(position, 1)) : float4(0, 0, 0, 1);
                return o;
            }

            fixed4 frag(v2f i) : SV_Target
            {
                if (i.shown < 0.5) discard;
                if (i.isDot > 0.5)
                {
                    // A white middle ringed in green, soft at its edge.
                    float r = length(i.at);
                    if (r > 1) discard;
                    float cover = 1 - smoothstep(0.82, 1, r);
                    float3 colour = lerp(_Core.rgb, _Color.rgb, smoothstep(0.45, 0.7, r));
                    return fixed4(colour * cover, cover);
                }
                // Brighter towards the screen and along its middle line, faint at the fingertip.
                float across = 1 - abs(i.at.y);
                float along = lerp(0.18, 1, smoothstep(0, 1, i.at.x));
                float3 colour = lerp(_Color.rgb, _Core.rgb, pow(across, 6) * 0.6) * across * along;
                return fixed4(colour, 0);
            }
            ENDCG
        }
    }
}
