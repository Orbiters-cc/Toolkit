// The orbit orb's light: a core bright in its middle that takes the colour towards its rim, and an additive halo around
// it, brightest where it is thickest and fading out at its edge so it has no outline. The halo material hides itself for
// viewers who hide custom shaders (its VRCFallback tag); the core falls back to an unlit colour.
// OrbitOrbInstaller copies this file into each orbit orb's folder as "Orbit orb.shader".
Shader "Orbiters/Orbit Orb"
{
    Properties
    {
        _Color ("Colour", Color) = (0.2, 0.75, 1, 1)
        _Intensity ("Intensity", Range(0, 4)) = 1.5
        _Falloff ("Edge falloff", Range(0.25, 8)) = 2
        [ToggleUI] _Halo ("Halo", Float) = 0
        [Enum(UnityEngine.Rendering.BlendMode)] _DstBlend ("Destination blend", Float) = 0
        [Enum(Off, 0, On, 1)] _ZWrite ("Depth write", Float) = 1
    }
    SubShader
    {
        Tags { "RenderType" = "Opaque" "Queue" = "Geometry" "VRCFallback" = "Unlit" }
        Pass
        {
            Blend One [_DstBlend]
            ZWrite [_ZWrite]
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile_instancing
            #include "UnityCG.cginc"

            fixed4 _Color;
            float _Intensity, _Falloff, _Halo;

            struct appdata { float4 vertex : POSITION; float3 normal : NORMAL; UNITY_VERTEX_INPUT_INSTANCE_ID };
            struct v2f { float4 pos : SV_POSITION; float3 normal : TEXCOORD0; float3 view : TEXCOORD1; UNITY_VERTEX_OUTPUT_STEREO };

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                o.pos = UnityObjectToClipPos(v.vertex);
                o.normal = UnityObjectToWorldNormal(v.normal);
                o.view = WorldSpaceViewDir(v.vertex);
                return o;
            }

            fixed4 frag(v2f i) : SV_Target
            {
                float facing = saturate(dot(normalize(i.normal), normalize(i.view)));
                // The halo only adds light.
                if (_Halo > .5) return fixed4(_Color.rgb * pow(facing, _Falloff) * _Intensity, 1);
                float rim = pow(1 - facing, _Falloff);
                return fixed4(lerp(lerp(_Color.rgb, 1, .65), _Color.rgb, rim) * _Intensity, 1);
            }
            ENDCG
        }
    }
    Fallback "Unlit/Color"
}
