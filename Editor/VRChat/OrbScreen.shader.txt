// N4N0's projected screen (Orbiters contracts, Docs/orb.md, Screen): N4N0's menu in front of its owner, the same pages
// Orbiters - N4N0's SteamVR overlay shows. A page's unchanging part is a picture (_Pages, one slice per page, drawn by
// the desktop from the menu's layout); what changes is drawn over it from the screen's memory: 744 bytes (1024 with
// 128-byte thumbnails) the animator keeps two to a material property (_W0 on, each b0 + 256 × b1), read here as one
// array. _Layout says, row by row for each page, where its text slots, thumbnails and buttons are (4-pixel units).
// Thumbnails are colour pictures that sharpen layer by layer as they arrive (Docs/orb.md, Thumbnails): their decoder is
// generated from the design table (OrbThumbnail.json) for N4N0's thumbnail size.
// It looks projected, as N4N0's speech panel (OrbText's Plate): the page's dark card is the projection's tinted,
// see-through plate (what the page shows stays solid), brighter towards its top, with faint scan lines running down, a thin
// bright line just inside its border and a soft glow past it (the mesh has a margin for it); N4N0's beam reaches its corners
// (OrbBeam, the screen's second part). The retro look keeps its own tube instead.
// Pages are 1024 × 576 pixels. Seen from behind (by the people around, or in a mirror) it shows the back, slice 0 of
// the pages ("N4N0 menu, look at the other side"): the menu is its owner's, on the side they face. With _Retro (the retro
// look, Docs/orb.md) the pages are the retro ones and what changes is drawn as Orbiters - N4N0's retro overlay draws it:
// cut corners, hover outlines, pressed elements inverted, chosen ones in brackets, pictures as green dots, scanlines.
// Viewers who hide custom shaders see nothing of it (VRCFallback). OrbInstaller writes this file, the word properties and the decoder
// filled in, as "N4N0 screen.shader" (or "N4N0 screen 128 B.shader").
Shader "Orbiters/N4N0 Screen"
{
    Properties
    {
        _Pages ("Pages", 2DArray) = "" {}
        _Layout ("Layout", 2D) = "black" {}
        _MainTex ("Glyph atlas", 2D) = "black" {}
        _Card ("Card", Color) = (0.063, 0.063, 0.071, 1)
        _Text ("Text", Color) = (0.925, 0.929, 0.933, 1)
        _Text2 ("Secondary text", Color) = (0.545, 0.545, 0.58, 1)
        _Accent ("Accent", Color) = (0, 0.855, 0.427, 1)
        _Placeholder ("Picture loading", Color) = (0.153, 0.153, 0.165, 1)
        [ToggleUI] _Retro ("Retro look", Float) = 0
        _Dim ("Retro dim", Color) = (0.094, 0.384, 0.196, 1)
        _Hot ("Retro hot", Color) = (0.784, 1, 0.843, 1)
        _Plate ("Projected plate", Color) = (0.008, 0.03, 0.02, 0.82)
        _Edge ("Projected edge", Color) = (0, 0.855, 0.427, 1)
        _Glow ("Projected glow", Range(0, 2)) = 1
        [HideInInspector] _Pictures ("Full pictures (CRT B)", 2D) = "black" {}
        [HideInInspector] _PictureData ("Picture data (CRT A, for the diagnostics)", 2D) = "black" {}
//@PROPERTIES
    }
    SubShader
    {
        // OrbScreen: the files' revision (OrbInstaller.ScreenRevision); a screen of an older one is swapped for these.
        Tags { "Queue" = "Transparent" "RenderType" = "Transparent" "IgnoreProjector" = "True" "VRCFallback" = "Hidden" "OrbScreen" = "17" }
        Cull Off
        ZWrite Off
        Blend SrcAlpha OneMinusSrcAlpha

        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma target 4.5
            #pragma multi_compile_instancing
            #include "UnityCG.cginc"
//@DEFINES

            UNITY_DECLARE_TEX2DARRAY(_Pages);
            Texture2D _Layout;
            sampler2D _MainTex;
            float4 _MainTex_TexelSize;
            fixed4 _Card, _Text, _Text2, _Accent, _Placeholder;
            float _Retro;
            fixed4 _Dim, _Hot, _Plate, _Edge;
            float _Glow;

            // The screen's memory as one array over its 372 properties (each takes a register of its own).
            CBUFFER_START(WordBuffer)
            float _Words[SCREEN_WORDS] : packoffset(c0);
//@BUFFER
            CBUFFER_END

            // Every word property used, so none is stripped (never added: _Card's alpha is never negative).
            float SumWords()
            {
                return 0
//@SUM
                ;
            }

            static const float2 Page = float2(1024, 576);
            // The screen's memory (Docs/orb.md, Screen).
            static const int Hover = 2, Pressed = 3, On = 4, Hidden = 8, Layers = 16, Unavailable = 21, TextStart = 24, PicturesStart = 384,
                PictureBytes = THUMBNAIL_BYTES;
            // The layout table's columns.
            static const int TableSlots = 1, TablePictures = 33, TableElements = 40, LayoutRows = 34;
            // Slice 0 is the back; every other slice is a page.
            static const int Pages = LayoutRows - 1;

            struct appdata
            {
                float4 vertex : POSITION;
                float2 uv : TEXCOORD0;
                UNITY_VERTEX_INPUT_INSTANCE_ID
            };

            struct v2f
            {
                float4 pos : SV_POSITION;
                float2 uv : TEXCOORD0;
                // 1 when the camera is behind the screen (the quad's front faces -z): a side that VFACE gets wrong in
                // mirrors and render textures.
                float behind : TEXCOORD1;
                UNITY_VERTEX_OUTPUT_STEREO
            };

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_OUTPUT(v2f, o);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                o.pos = UnityObjectToClipPos(v.vertex);
                o.uv = v.uv;
                float3 world = mul(unity_ObjectToWorld, v.vertex).xyz;
                o.behind = dot(UnityObjectToWorldNormal(float3(0, 0, -1)), _WorldSpaceCameraPos - world) < 0;
                return o;
            }

            // Byte i of the memory: word i / 2 holds bytes 2k (low) and 2k + 1 (high) as b0 + 256 b1.
            int Byte(int i)
            {
                float w = _Words[clamp(i >> 1, 0, SCREEN_WORDS - 1)];
                float high = floor((w + .5) / 256);
                return (int)((i & 1) ? high : round(w - high * 256));
            }

            bool Bit(int at, int index) { return ((Byte(at + index / 7) >> (index % 7)) & 1) == 1; }

//@THUMBNAIL

            // Full pictures (Docs/orb-picture.md): CRT B's pictures, signatures and heartbeat (the contracts' include).
//@PICTURES

            // The full picture a thumbnail place shows instead of its thumbnail: its slot (results 0 to 4, the music's art 5,
            // the card's picture 6) when it holds the picture whose signature is the thumbnail's first 8 bytes, CRT B ran and
            // a screen pixel spans under 2 of its texels (smaller, the smooth thumbnail looks better); else why not (negative,
            // PictureWhyColour): -1 B never ran or no thumbnail yet, -2 too small, -3 another picture, -4 none.
            int FullPicture(int page, int t, int layers, float width, float pixel, bool alive)
            {
                int slot = page == 11 ? 5 : (page == 15 ? 6 : t);
                if (!alive || layers <= 0) return -1;
                if (PictureTexelsPerPixel(slot, width, pixel) >= 2) return -2;
                int at = PicturesStart + t * PictureBytes;
                uint4 a = uint4(Byte(at), Byte(at + 1), Byte(at + 2), Byte(at + 3)), b = uint4(Byte(at + 4), Byte(at + 5), Byte(at + 6), Byte(at + 7));
                int match = PictureMatch(slot, a, b);
                return match == 1 ? slot : (match == 0 ? -3 : -4);
            }

            // A texel of the layout table, its bytes (the table's rows run from page 0 at the top of its picture).
            int4 Texel(int column, int page) { return (int4)round(_Layout.Load(int3(column, LayoutRows - 1 - page, 0)) * 255); }

            // The signed distance from a rounded rectangle (x, y, width, height) at p (negative inside), and its coverage, a
            // pixel's width antialiased.
            float RoundedDistance(float2 p, float4 r, float radius)
            {
                float2 half = r.zw * .5;
                float2 d = abs(p - (r.xy + half)) - (half - radius);
                return length(max(d, 0)) + min(max(d.x, d.y), 0) - radius;
            }

            float Rounded(float2 p, float4 r, float radius)
            {
                float pixel = max(fwidth(p.x), 1e-4);
                return saturate(.5 - RoundedDistance(p, r, radius) / pixel);
            }

            // The retro look's shapes, in page pixels: a rectangle (x, y, width, height) with its corners cut by `cut`, as a
            // signed distance (negative inside); coverage of a distance a pixel wide; corner brackets of arms `len` long and
            // `width` thick on a rectangle's corners.
            float Chamfer(float2 p, float4 r, float cut)
            {
                float2 extent = r.zw * .5;
                float2 q = abs(p - (r.xy + extent));
                float box = max(q.x - extent.x, q.y - extent.y);
                return max(box, (q.x + q.y - (extent.x + extent.y - cut)) * .70710678);
            }
            float Cover(float distance, float pixel) { return saturate(.5 - distance / pixel); }
            float Brackets(float2 p, float4 r, float len, float width, float pixel)
            {
                float2 lo = r.xy, hi = r.xy + r.zw, mid = (lo + hi) * .5;
                float2 corner = float2(p.x < mid.x ? lo.x : hi.x, p.y < mid.y ? lo.y : hi.y);
                float2 q = (p - corner) * float2(p.x < mid.x ? 1 : -1, p.y < mid.y ? 1 : -1);
                float across = max(max(-q.x, q.x - len), max(-q.y, q.y - width));
                float down = max(max(-q.x, q.x - width), max(-q.y, q.y - len));
                return Cover(min(across, down), pixel);
            }
            // Retro pictures (Docs/orb.md, Retro look): 96 × 54 dots, the light stretched and dithered (4 × 4 Bayer) into
            // four steps.
            static const float2 Dots = float2(96, 54);
            static const float Bayer[16] = { 0, 8, 2, 10, 12, 4, 14, 6, 3, 11, 1, 9, 15, 7, 13, 5 };
            int DotStep(float light, float2 cell)
            {
                float level = saturate((light - .08) / .84) * 3;
                float threshold = (Bayer[(int)fmod(cell.y, 4) * 4 + (int)fmod(cell.x, 4)] + .5) / 16;
                return min((int)floor(level + threshold), 3);
            }

            // The back, slice 0 of the pages (N4N0 over "N4N0 menu" and "look at the other side"), the right way round from
            // behind.
            fixed4 Back(float2 uv)
            {
                uv.x = 1 - uv.x;
                float2 p = float2(uv.x, 1 - uv.y) * Page;
                float alpha = _Retro > .5 ? Cover(Chamfer(p, float4(0, 0, Page), 20), max(fwidth(p.x), 1e-4)) : Rounded(p, float4(0, 0, Page), 28);
                if (alpha <= 0) discard;
                return fixed4(UNITY_SAMPLE_TEX2DARRAY(_Pages, float3(uv, 0)).rgb, alpha);
            }

            // The retro look's tube: every third row darker (averaged once rows get thinner than a pixel, `rowWidth` the rows
            // a pixel spans, so far away they never shimmer), the corners faded.
            fixed3 Retro(fixed3 colour, float2 uv, float rowWidth)
            {
                float3 c = colour;
                float row = (1 - uv.y) * Page.y;
                // Rows 2, 5, 8… (fmod's remainder is not exact on GPUs: a third of the row index's fraction instead).
                float scan = frac(floor(row) / 3) > .5 ? .72 : 1;
                c *= lerp(scan, 1 - .28 / 3, saturate(rowWidth - .5));
                float2 d = uv * 2 - 1;
                return c * (1 - .22 * (.6 * d.x * d.x + d.y * d.y));
            }

            // The projected look over a side as drawn (`picture`: its page picture before anything was drawn over it; not
            // past the border, which is the glow). Where the picture is its dark card and nothing was drawn over it, the
            // plate shows, tinted and see-through; scan lines about five page rows apart run down over everything, gone once
            // they would be thinner than a pixel (`rowWidth`).
            fixed4 Projected(fixed4 c, fixed3 picture, float2 uv, float rowWidth)
            {
                float2 p = float2(uv.x, 1 - uv.y) * Page;
                // Hidden elements are covered with the card's own colour: that is plate too.
                float card = max((1 - saturate((Luminance(picture) - Luminance(_Card.rgb)) * 60)) * (1 - saturate(length(c.rgb - picture) * 20)),
                    1 - saturate(length(c.rgb - _Card.rgb) * 150));
                float shade = 1.2 - .35 * saturate(p.y / Page.y);
                float scan = lerp(1, .93 + .07 * sin(p.y * 1.3125 - _Time.y * 5), saturate(2.5 - rowWidth * 1.3125));
                c.rgb = lerp(c.rgb, _Plate.rgb * shade, card) * scan;
                c.a *= lerp(1, _Plate.a, card);
                // The thin bright line, two pixels wide and as far inside the border.
                float pixel = max(fwidth(p.x), 1e-4);
                float rule = 1 - smoothstep(1 - pixel, 1 + pixel, abs(RoundedDistance(p, float4(0, 0, Page), 28) + 2.4));
                c.rgb = lerp(c.rgb, _Edge.rgb * 1.4, rule * .85);
                c.a = max(c.a, rule * .85);
                return c;
            }

            fixed4 Front(v2f i);

            fixed4 frag(v2f i) : SV_Target
            {
                // Before any branch or discard: the page rows a pixel spans, for the scan lines.
                float rowWidth = fwidth(i.uv.y * Page.y);
                bool retro = _Retro > .5;
                // Past the page's border, in the mesh's margin, on either side: only the projection's glow.
                float border = RoundedDistance(float2(i.uv.x, 1 - i.uv.y) * Page, float4(0, 0, Page), 28);
                if (!retro && border > 0)
                {
                    float glow = exp(-border / 3.5) * .35 * _Glow;
                    if (glow < .003) discard;
                    return fixed4(_Edge.rgb, glow);
                }
                // The side the camera is on (the whole quad's: the branch is the same for every pixel of it).
                fixed4 c;
                if (i.behind > .5)
                {
                    c = Back(i.uv);
                    if (!retro) c = Projected(c, c.rgb, float2(1 - i.uv.x, i.uv.y), rowWidth);
                }
                else
                {
                    c = Front(i);
                    int page = Byte(0);
                    if (!retro) c = Projected(c, page >= 1 && page <= Pages ? UNITY_SAMPLE_TEX2DARRAY(_Pages, float3(i.uv, page)).rgb : _Card.rgb, i.uv, rowWidth);
                }
                if (retro) c.rgb = Retro(c.rgb, i.uv, rowWidth);
                return c;
            }

            fixed4 Front(v2f i)
            {
                float2 uv = i.uv;
                float2 p = float2(uv.x, 1 - uv.y) * Page;
                bool retro = _Retro > .5;
                float pixel = max(fwidth(p.x), 1e-4);
                float alpha = retro ? Cover(Chamfer(p, float4(0, 0, Page), 20), pixel) : Rounded(p, float4(0, 0, Page), 28);
                if (alpha <= 0) discard;
                // Retro: the pressed element, inverted once its text is drawn.
                float4 pressedAt = 0;
                float pressedCut = 0;
                int page = Byte(0);
                bool known = page >= 1 && page <= Pages;
                fixed3 colour = known ? UNITY_SAMPLE_TEX2DARRAY(_Pages, float3(uv.x, uv.y, page)).rgb : _Card.rgb;
                if (_Card.a < 0) colour.r += SumWords();
                if (!known) return fixed4(colour, alpha);
                int4 head = Texel(0, page);
                int hover = Byte(Hover) - 1, pressed = Byte(Pressed) - 1;

                // Buttons: hidden ones go, greyed out ones fade into the card, then what is hovered, pressed and switched on.
                [loop] for (int e = 0; e < head.w; e++)
                {
                    bool hidden = Bit(Hidden, e), on = e < 28 && Bit(On, e), unavailable = e < 21 && Bit(Unavailable, e);
                    if (!hidden && !on && !unavailable && e != hover && e != pressed) continue;
                    int4 at = Texel(TableElements + 2 * e, page);
                    int4 look = Texel(TableElements + 2 * e + 1, page);
                    float4 r = float4(at) * 4;
                    if (retro)
                    {
                        float cut = r.w < 64 ? 7 : 14;
                        if (hidden) colour = lerp(colour, _Card.rgb, Cover(Chamfer(p, r + float4(-3, -3, 6, 6), cut + 2), pixel));
                        else if (unavailable) colour = lerp(colour, _Card.rgb, .6 * Cover(Chamfer(p, r, cut), pixel));
                        else
                        {
                            if (e == pressed) { pressedAt = r; pressedCut = cut; }
                            else if (e == hover)
                            {
                                float d = Chamfer(p, r + float4(-3, -3, 6, 6), cut + 1);
                                colour = lerp(colour, _Accent.rgb, .1 * Cover(d, pixel));
                                colour = lerp(colour, _Hot.rgb, Cover(abs(d + 1.25) - 1.25, pixel));
                            }
                            if (on) colour = lerp(colour, _Hot.rgb, Brackets(p, r + float4(-6, -6, 12, 12), 14, 3, pixel));
                        }
                        continue;
                    }
                    // A hidden button goes, its soft edge too.
                    if (hidden)
                    {
                        colour = lerp(colour, _Card.rgb, Rounded(p, r + float4(-2, -2, 4, 4), look.x + 2));
                        continue;
                    }
                    float inside = Rounded(p, r, look.x);
                    if (inside <= 0) continue;
                    // Not available now (fast walk in desktop mode): the card shows through, as on the overlay.
                    if (unavailable)
                    {
                        colour = lerp(colour, _Card.rgb, .6 * inside);
                        continue;
                    }
                    // Looks 3 and 9: green and red buttons, darkened when pressed. White over the page in sRGB, as the
                    // overlay paints it.
                    bool solid = look.y == 3 || look.y == 9;
                    #ifdef UNITY_COLORSPACE_GAMMA
                    float3 g = colour;
                    #else
                    float3 g = LinearToGammaSpace(colour);
                    #endif
                    if (e == pressed) g = solid ? lerp(g, 0, .16 * inside) : lerp(g, 1, .12 * inside);
                    else if (e == hover) g = solid ? lerp(g, 1, .14 * inside) : lerp(g, 1, .06 * inside);
                    #ifdef UNITY_COLORSPACE_GAMMA
                    colour = g;
                    #else
                    colour = GammaToLinearSpace(g);
                    #endif
                    if (on)
                    {
                        float ring = inside * (1 - Rounded(p, r + float4(3, 3, -6, -6), max(look.x - 3, 0)));
                        colour = lerp(colour, _Accent.rgb, ring);
                    }
                }

                // Text: each slot's cells, a glyph of the atlas each (tiles twice as tall as wide, glyph g at column g mod 16,
                // row g div 16 from the top).
                [loop] for (int s = 0; s < head.y; s++)
                {
                    int4 slot = Texel(TableSlots + 2 * s, page);
                    float2 origin = float2(slot.xy) * 4;
                    float cell = slot.w;
                    float2 local = p - origin;
                    if (local.x < 0 || local.y < 0 || local.x >= slot.z * cell || local.y >= 2 * cell) continue;
                    int4 more = Texel(TableSlots + 2 * s + 1, page);
                    int column = (int)(local.x / cell);
                    int glyph = Byte(TextStart + more.x + 256 * more.y + column);
                    if (glyph == 0) continue;
                    float2 inCell = float2(local.x / cell - column, local.y / (2 * cell));
                    float2 atlas = float2((glyph % 16 + inCell.x) / 16, 1 - (glyph / 16 + inCell.y) / 16);
                    // Gradients from the screen position, smooth across tiles (a tile's edge is no jump in the mip level).
                    float2 dx = ddx(p) / float2(16 * cell, -32 * cell), dy = ddy(p) / float2(16 * cell, -32 * cell);
                    float coverage = tex2Dgrad(_MainTex, atlas, dx, dy).r;
                    int style = more.z & 15;
                    fixed3 ink = style == 2 ? _Accent.rgb : style == 1 ? _Text2.rgb : _Text.rgb;
                    colour = lerp(colour, ink, coverage);
                }

                // Thumbnails (results): colour pictures, the layers received so far summed at this point of the picture; the
                // full picture instead once it arrived.
                bool alive = PicturesAlive();
                // The full pictures' diagnostics (the flags' bit 3, Orbiters - N4N0's Picture diagnostics): each place framed
                // in the colour of why it shows what it shows, and the CRTs' state in a row at the bottom left.
                bool diagnostics = (Byte(1) & 8) != 0;
                [loop] for (int t = 0; t < head.z; t++)
                {
                    float4 r = float4(Texel(TablePictures + t, page)) * 4;
                    int layers = Byte(Layers + t);
                    int full = FullPicture(page, t, layers, r.z, pixel, alive);
                    if (diagnostics) colour = lerp(colour, PictureWhyColour(full), saturate(1.5 - abs(RoundedDistance(p, r, 10) - 3)));
                    if (retro)
                    {
                        // The music page's art place is part of its picture until the art came.
                        if (layers == 0 && page == 11) continue;
                        float box = Cover(Chamfer(p, r, 0), pixel);
                        if (layers > 0) colour = lerp(colour, _Accent.rgb, Brackets(p, r + float4(-3, -3, 6, 6), 10, 1.5, pixel));
                        if (box <= 0) continue;
                        fixed3 shade = _Placeholder.rgb;
                        if (layers > 0)
                        {
                            float2 f = saturate((p - r.xy) / r.zw) * Dots;
                            float2 cell = min(floor(f), Dots - 1);
                            float light = full >= 0 ? dot(PictureColour(full, (cell + .5) / Dots), float3(.299, .587, .114))
                                : Thumbnail(PicturesStart + t * PictureBytes, (cell + .5) / Dots, layers).x;
                            int level = DotStep(light, cell);
                            shade = level == 3 ? _Hot.rgb : level == 2 ? _Accent.rgb : level == 1 ? _Dim.rgb : _Card.rgb;
                            // A dark gap between dots once they are large enough to show one.
                            float2 inCell = f - cell;
                            if (r.z / Dots.x > 3 * pixel && max(inCell.x, inCell.y) > .78) shade = _Card.rgb;
                        }
                        colour = lerp(colour, shade, box);
                        continue;
                    }
                    float inside = Rounded(p, r, 10);
                    if (inside <= 0) continue;
                    // The music page's art place is part of its picture until the art came.
                    if (layers == 0 && page == 11) continue;
                    fixed3 shade = _Placeholder.rgb;
                    if (layers > 0)
                    {
                        float2 inPicture = saturate((p - r.xy) / r.zw);
                        shade = full >= 0 ? PictureColour(full, inPicture) : YccToRgb(Thumbnail(PicturesStart + t * PictureBytes, inPicture, layers));
                        #ifndef UNITY_COLORSPACE_GAMMA
                        shade = GammaToLinearSpace(shade);
                        #endif
                    }
                    colour = lerp(colour, shade, inside);
                }

                // N4N0 listening (Home, four, six or seven tiles): its microphone breathes.
                if ((page == 1 || page == 10 || page == 13) && (Byte(1) & 1))
                {
                    float pulse = .5 + .5 * sin(_Time.y * 4);
                    float d = abs(length(p - float2(60, 52)) - (31 + 4 * pulse));
                    colour = lerp(colour, _Accent.rgb, saturate(1.5 - d) * (.7 - .5 * pulse));
                }
                // Retro: the pressed element lit, what is on it dark.
                if (retro && pressedAt.z > 0)
                    colour = lerp(colour, _Accent.rgb + _Card.rgb - colour, Cover(Chamfer(p, pressedAt, pressedCut), pixel));
                if (diagnostics)
                {
                    float4 d = PictureDiagnostics(p);
                    colour = lerp(colour, d.rgb, d.a);
                }
                return fixed4(colour, alpha);
            }
            ENDCG
        }
    }
}
