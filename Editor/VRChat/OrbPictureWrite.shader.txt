// N4N0's full pictures, the bridge's first pass (Orbiters contracts, Docs/orb-picture.md): keeps the frame as it was (a
// named grab, once for every wearer), then draws this frame's picture chunks (the values N4N0's animator copies onto this
// material: the synced gate and values, the owner's local values) as pixels in the wearer's slot, straight in clip space,
// so the same pixels land in every camera and, stereo instanced, in each eye. CRT A finds them in the frame's grab (the
// restore pass's), and the restore pass puts back what was under the slot. Viewers who hide custom shaders draw none of it.
// OrbInstaller writes this file with the include and the values' properties written in.
Shader "Orbiters/N4N0 Picture Bridge Write"
{
    Properties
    {
        [HideInInspector] _PictureData ("Picture data (CRT A)", 2D) = "black" {}
        [HideInInspector] _Pictures ("Pictures (CRT B)", 2D) = "black" {}
//@PROPERTIES
    }
    SubShader
    {
        Tags { "Queue" = "Overlay+10" "RenderType" = "Overlay" "IgnoreProjector" = "True" "DisableBatching" = "True" "VRCFallback" = "Hidden" "PreviewType" = "Plane" }
        GrabPass { "_N4N0PictureOriginal" }
        Pass
        {
            ZTest Always ZWrite Off Cull Off Blend Off
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma target 4.5
            #pragma multi_compile_instancing
            #include "UnityCG.cginc"
            #define PICTURE_BRIDGE
            #include "OrbitersPicture.cginc"

            struct appdata { float4 vertex : POSITION; float2 uv : TEXCOORD0; UNITY_VERTEX_INPUT_INSTANCE_ID };
            struct v2f { float4 pos : SV_Position; UNITY_VERTEX_OUTPUT_STEREO };

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_OUTPUT(v2f, o);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                o.pos = PicBridgeClip(v.uv);
                return o;
            }

            float4 frag(v2f i) : SV_Target
            {
                UNITY_SETUP_STEREO_EYE_INDEX_POST_VERTEX(i);
                return PicBridgeWrite(i.pos.xy);
            }
            ENDCG
        }
    }
}
