// The avatar gun's shot (Orbiters contracts, Docs/avatar-gun.md): an additive beam from the muzzle to where the gun's ray
// ends, as N4N0's beam glows (N4N0Beam): a thin white core in a green glow, pulses running out along it, a star-shaped
// flash at the muzzle and a burst where the ray ends. One skinned mesh: the beam's near end rides the muzzle, its far end
// the ray's end. The first UV is where a vertex is across and along its strip (or on its flash's square), the second's x
// which part it is (0 the beam, 1 the muzzle's flash, 2 the end's). _Fire (1 when the shot leaves, the animator fades it
// to 0) lights it all. Viewers who hide custom shaders see nothing of it (VRCFallback).
// AvatarGunInstaller copies this file as "Avatar gun beam.shader".
Shader "Orbiters/Avatar gun beam"
{
    Properties
    {
        _Color ("Glow", Color) = (0, 0.855, 0.427, 1)
        _Fire ("Fire", Range(0, 1)) = 0
    }
    SubShader
    {
        Tags { "Queue" = "Transparent" "RenderType" = "Transparent" "IgnoreProjector" = "True" "VRCFallback" = "Hidden" }
        Cull Off
        ZWrite Off
        Blend One One

        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile_instancing
            #include "UnityCG.cginc"

            fixed4 _Color;
            float _Fire;

            struct appdata
            {
                float4 vertex : POSITION;
                float2 uv : TEXCOORD0;
                float2 part : TEXCOORD1;
                UNITY_VERTEX_INPUT_INSTANCE_ID
            };

            struct v2f
            {
                float4 pos : SV_POSITION;
                float2 uv : TEXCOORD0;
                float part : TEXCOORD1;
                UNITY_VERTEX_OUTPUT_STEREO
            };

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_OUTPUT(v2f, o);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                // Nothing to draw between shots.
                o.pos = _Fire > 0.001 ? UnityObjectToClipPos(v.vertex) : float4(0, 0, -1, 1);
                o.uv = v.uv;
                o.part = v.part.x;
                return o;
            }

            fixed4 frag(v2f i) : SV_Target
            {
                float3 glow = _Color.rgb;
                if (i.part < 0.5)
                {
                    float across = abs(i.uv.x - 0.5) * 2;
                    float core = exp(-across * across * 90);
                    float halo = exp(-across * across * 7) * 0.45;
                    float along = i.uv.y;
                    // Pulses running out from the muzzle, and a beam that thins out towards its far end.
                    float run = 0.7 + 0.3 * sin(along * 24 - _Time.y * 30);
                    float fade = (1 - 0.55 * along) * smoothstep(0, 0.004, along);
                    float3 colour = lerp(glow, 1, core) * (core * 1.6 + halo * run);
                    return fixed4(colour * fade * _Fire, 1);
                }
                float2 p = i.uv * 2 - 1;
                float r = length(p);
                float disc = exp(-r * r * 9);
                // A four-pointed star along the square's axes.
                float star = exp(-abs(p.x) * 28) * exp(-abs(p.y) * 2.5) + exp(-abs(p.y) * 28) * exp(-abs(p.x) * 2.5);
                float shape = saturate(disc + star * 0.8) * saturate(1 - r);
                // The muzzle's flash is over sooner than the end's burst.
                float strength = i.part < 1.5 ? _Fire * _Fire * 1.4 : _Fire;
                float3 colour = lerp(glow, 1, exp(-r * r * 30));
                return fixed4(colour * shape * strength, 1);
            }
            ENDCG
        }
    }
}
