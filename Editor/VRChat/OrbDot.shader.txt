// N4N0's pointer (Orbiters contracts, Docs/orb.md, Looking and clicking, and Pointing): where its owner points at N4N0's
// menu, a plain white dot ringed in Orbiters' green, unlit, the same from every side (a sphere's middle white, its rim
// green). Over everything when it marks a point (_ZTest Always: desktop mode's dot before the eyes, a ray's bead on the
// screen); in the scene when it is a ray's beam. Only its owner sees it: N4N0's animator shows it on their client alone.
Shader "Orbiters/N4N0 Dot"
{
    Properties
    {
        _Color ("Colour", Color) = (1, 1, 1, 1)
        _Rim ("Rim", Color) = (0, 0.855, 0.427, 1)
        [Enum(UnityEngine.Rendering.CompareFunction)] _ZTest ("Depth test", Float) = 8
    }
    SubShader
    {
        Tags { "Queue" = "Overlay" "RenderType" = "Transparent" "IgnoreProjector" = "True" "VRCFallback" = "Hidden" }
        Pass
        {
            ZWrite Off
            ZTest [_ZTest]
            Cull Back
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile_instancing
            #include "UnityCG.cginc"

            fixed4 _Color, _Rim;

            struct appdata
            {
                float4 vertex : POSITION;
                float3 normal : NORMAL;
                UNITY_VERTEX_INPUT_INSTANCE_ID
            };

            struct v2f
            {
                float4 pos : SV_POSITION;
                float facing : TEXCOORD0;
                UNITY_VERTEX_OUTPUT_STEREO
            };

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_OUTPUT(v2f, o);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                o.pos = UnityObjectToClipPos(v.vertex);
                float3 world = mul(unity_ObjectToWorld, v.vertex).xyz;
                o.facing = saturate(dot(normalize(UnityObjectToWorldNormal(v.normal)), normalize(_WorldSpaceCameraPos - world)));
                return o;
            }

            fixed4 frag(v2f i) : SV_Target
            {
                // The rim as the sphere turns away: a ring of a fifth of the dot's width.
                float ring = smoothstep(0.42, 0.62, 1 - i.facing);
                return fixed4(lerp(_Color.rgb, _Rim.rgb, ring), 1);
            }
            ENDCG
        }
    }
}
