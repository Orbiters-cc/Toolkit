// Contract: Docs/capture-strip.md and Docs/depth-frames.md. CPU twins: CSharp/CaptureStrip.cs, CSharp/DepthFrames.cs,
// Rust~/src/strip.rs, Rust~/src/depth.rs. Any change here must keep TestVectors~ decoding on both sides.
//
// Include after UnityCG.cginc. Needs shader model 4.0 or higher (integer maths, asuint).
// Define ORBITERS_DEPTH_TEXTURE_DECLARED if your shader already declares _CameraDepthTexture, and
// ORBITERS_VRCHAT_GLOBALS_DECLARED if it already declares _VRChatCameraMode / _VRChatMirrorMode.

#ifndef ORBITERS_CAPTURE_INCLUDED
#define ORBITERS_CAPTURE_INCLUDED

#ifndef ORBITERS_DEPTH_TEXTURE_DECLARED
UNITY_DECLARE_DEPTH_TEXTURE(_CameraDepthTexture);
#endif

#ifndef ORBITERS_VRCHAT_GLOBALS_DECLARED
float _VRChatCameraMode;
float _VRChatMirrorMode;
#endif

// Material inputs. Leaving them at 0 gives the VRChat avatar behaviour.
float _OrbitersUseTimeOverride; // 1: _OrbitersTimeOverride replaces _Time.y (the simulator renders several frames per editor frame)
float _OrbitersTimeOverride;
float _OrbitersSource;          // flags source field: 0 avatar, 1 simulator, 2 test
float _OrbitersEye;             // 0 mono/left, 1 right
float _OrbitersFace;            // 0 none, 1..6 cube face
float _OrbitersDepthEvery;      // capture kit: about one frame in N is a depth frame (0: none)

#define ORBITERS_WORD0 0x4F420002u
#define ORBITERS_PAYLOAD_BITS 736u
#define ORBITERS_CODED_BITS 2208u
#define ORBITERS_BIT_LEVEL 0.3

static const uint ORBITERS_CRC_NIBBLE[16] =
{
    0x00000000u, 0x1DB71064u, 0x3B6E20C8u, 0x26D930ACu, 0x76DC4190u, 0x6B6B51F4u, 0x4DB26158u, 0x5005713Cu,
    0xEDB88320u, 0xF00F9344u, 0xD6D6A3E8u, 0xCB61B38Cu, 0x9B64C2B0u, 0x86D3D2D4u, 0xA00AE278u, 0xBDBDF21Cu
};

static const float3 ORBITERS_PATCH[6] =
{
    float3(0.05, 0.05, 0.05), float3(0.18, 0.18, 0.18), float3(0.5, 0.5, 0.5),
    float3(0.5, 0.0, 0.0), float3(0.0, 0.5, 0.0), float3(0.0, 0.0, 0.5)
};

struct OrbitersStripLayout
{
    bool valid;
    uint width, height, block, cols, groups, rows, x0, y0, patchSize, patchX0, patchY;
    uint4 reserved; // x0, y0, x1, y1 (half-open)
};

OrbitersStripLayout OrbitersComputeStripLayout(uint width, uint height)
{
    OrbitersStripLayout l = (OrbitersStripLayout)0;
    l.width = width;
    l.height = height;
    l.block = max(4u, height / 135u);
    l.cols = (width * 7u / 10u) / l.block / 16u * 16u;
    l.groups = l.cols / 16u;
    uint perRow = max(1u, l.groups * 14u);
    l.rows = (ORBITERS_CODED_BITS + perRow - 1u) / perRow;
    l.x0 = (width - l.cols * l.block) / 2u;
    l.y0 = (height * 8u / 100u) / l.block * l.block;
    l.patchSize = 2u * l.block;
    uint patchTotal = 6u * l.patchSize + 5u * l.block;
    l.patchX0 = (width - patchTotal) / 2u;
    l.patchY = l.y0 + l.rows * l.block + l.block;
    uint bottom = l.patchY + l.patchSize + l.block;
    l.valid = l.cols >= 32u && patchTotal + 2u * l.block <= width && bottom <= height;
    uint rx0 = min(l.x0, l.patchX0);
    l.reserved = uint4(rx0 > l.block ? rx0 - l.block : 0u,
                       l.y0 > l.block ? l.y0 - l.block : 0u,
                       min(width, max(l.x0 + l.cols * l.block, l.patchX0 + patchTotal) + l.block),
                       bottom);
    return l;
}

// Pixel coordinates with row 0 at the top of the final picture, whatever the render target orientation.
uint2 OrbitersTopDownPixel(float4 svPosition)
{
    uint2 p = uint2(svPosition.xy);
    if (_ProjectionParams.x < 0)
        p.y = (uint)_ScreenParams.y - 1u - p.y;
    return p;
}

float OrbitersTime()
{
    return _OrbitersUseTimeOverride > 0.5 ? _OrbitersTimeOverride : _Time.y;
}

// lowbias32 (Chris Wellons' integer hash).
uint OrbitersHash(uint x)
{
    x ^= x >> 16;
    x *= 0x7FEB352Du;
    x ^= x >> 15;
    x *= 0x846CA68Bu;
    x ^= x >> 16;
    return x;
}

// Whether the capture kit makes this frame a depth frame: picked from the frame's game time, so every draw of the
// frame agrees without any state (about one frame in _OrbitersDepthEvery, at random).
bool OrbitersKitDepthFrame()
{
    uint every = (uint)max(_OrbitersDepthEvery, 0.0);
    return every > 0u && OrbitersHash(asuint(OrbitersTime())) % every == 0u;
}

uint OrbitersFlags(bool depthFrame)
{
    uint kind = depthFrame ? 1u : 0u;
    uint cameraMode = (uint)clamp(_VRChatCameraMode, 0.0, 7.0);
    uint mirror = _VRChatMirrorMode > 0.5 ? 1u : 0u;
    return kind | (cameraMode << 2) | (mirror << 5) | (((uint)_OrbitersEye & 3u) << 6) |
           (((uint)_OrbitersFace & 15u) << 8) | (((uint)_OrbitersSource & 15u) << 12);
}

// Bits of unity_CameraToWorld[k / 4][k % 4]. Float components are only read at static indices and picked with integer
// selects: FXC compiles a dynamic float component index (row[k & 3]) as a dot product with a one-hot vector, which
// turns -0.0 into +0.0, so the blocks lost the sign of zero while the CRC loop kept it.
uint OrbitersCameraToWorldBits(uint k)
{
    uint4 r0 = asuint(unity_CameraToWorld[0]);
    uint4 r1 = asuint(unity_CameraToWorld[1]);
    uint4 r2 = asuint(unity_CameraToWorld[2]);
    uint r = k >> 2, c = k & 3u;
    uint4 row = r == 0u ? r0 : (r == 1u ? r1 : r2);
    return c == 0u ? row.x : (c == 1u ? row.y : (c == 2u ? row.z : row.w));
}

uint OrbitersPayloadWord(uint i, uint flags, uint width, uint height)
{
    uint word = ((width & 0xFFFFu) << 16) | (height & 0xFFFFu); // word 21
    if (i == 0u) word = ORBITERS_WORD0;
    else if (i == 1u) word = flags;
    else if (i == 2u) word = asuint(OrbitersTime());
    else if (i < 15u) word = OrbitersCameraToWorldBits(i - 3u);
    else if (i == 15u) word = asuint(unity_CameraProjection[0][0]);
    else if (i == 16u) word = asuint(unity_CameraProjection[1][1]);
    else if (i == 17u) word = asuint(unity_CameraProjection[0][2]);
    else if (i == 18u) word = asuint(unity_CameraProjection[1][2]);
    else if (i == 19u) word = asuint(_ProjectionParams.y);
    else if (i == 20u) word = asuint(_ProjectionParams.z);
    return word;
}

uint OrbitersCrcStep(uint crc, uint byteValue)
{
    crc ^= byteValue;
    crc = ORBITERS_CRC_NIBBLE[crc & 15u] ^ (crc >> 4);
    crc = ORBITERS_CRC_NIBBLE[crc & 15u] ^ (crc >> 4);
    return crc;
}

uint OrbitersWord(uint i, uint flags, uint width, uint height)
{
    uint word;
    if (i < 22u)
    {
        word = OrbitersPayloadWord(i, flags, width, height);
    }
    else
    {
        uint crc = 0xFFFFFFFFu;
        [loop] for (uint j = 0u; j < 22u; j++)
        {
            uint w = OrbitersPayloadWord(j, flags, width, height);
            crc = OrbitersCrcStep(crc, w & 0xFFu);
            crc = OrbitersCrcStep(crc, (w >> 8) & 0xFFu);
            crc = OrbitersCrcStep(crc, (w >> 16) & 0xFFu);
            crc = OrbitersCrcStep(crc, w >> 24);
        }
        word = crc ^ 0xFFFFFFFFu;
    }
    return word;
}

#define ORBITERS_STRIP_NONE 0u
#define ORBITERS_STRIP_FIXED 1u
#define ORBITERS_STRIP_BIT 2u

// What pixel p (top-down) shows of a valid strip layout: ORBITERS_STRIP_NONE outside the strip and its patches,
// ORBITERS_STRIP_FIXED a reference block, an unused block or a patch (linear colour in color), ORBITERS_STRIP_BIT a
// payload bit (its index in bitIndex).
uint OrbitersStripBlock(uint2 p, OrbitersStripLayout l, out float3 color, out uint bitIndex)
{
    // One exit: FXC takes early returns with out parameters for uses of uninitialised variables.
    color = 0;
    bitIndex = 0u;
    uint what = ORBITERS_STRIP_NONE;
    if (p.x >= l.x0 && p.y >= l.y0)
    {
        uint col = (p.x - l.x0) / l.block;
        uint row = (p.y - l.y0) / l.block;
        if (col < l.cols && row < l.rows)
        {
            uint g = col % 16u;
            what = ORBITERS_STRIP_FIXED;
            if (g == 1u)
            {
                color = ORBITERS_BIT_LEVEL;
            }
            else if (g >= 2u)
            {
                uint d = row * l.groups * 14u + (col / 16u) * 14u + g - 2u;
                if (d < ORBITERS_CODED_BITS)
                {
                    bitIndex = d % ORBITERS_PAYLOAD_BITS;
                    what = ORBITERS_STRIP_BIT;
                }
            }
        }
    }
    if (what == ORBITERS_STRIP_NONE && p.y >= l.patchY && p.y < l.patchY + l.patchSize && p.x >= l.patchX0)
    {
        uint rel = p.x - l.patchX0;
        uint stride = l.patchSize + l.block;
        uint index = rel / stride;
        if (index < 6u && rel % stride < l.patchSize)
        {
            color = ORBITERS_PATCH[index];
            what = ORBITERS_STRIP_FIXED;
        }
    }
    return what;
}

// Strip and reference patches. Returns true when pixel p (top-down) belongs to them; color is linear.
bool OrbitersStripPixel(uint2 p, uint width, uint height, bool depthFrame, out float3 color)
{
    color = 0;
    OrbitersStripLayout l = OrbitersComputeStripLayout(width, height);
    if (!l.valid) return false;
    uint bitIndex = 0u;
    uint what = OrbitersStripBlock(p, l, color, bitIndex);
    if (what == ORBITERS_STRIP_BIT)
    {
        uint word = OrbitersWord(bitIndex / 32u, OrbitersFlags(depthFrame), width, height);
        color = ((word >> (31u - bitIndex % 32u)) & 1u) != 0u ? ORBITERS_BIT_LEVEL : 0.0;
    }
    return what != ORBITERS_STRIP_NONE;
}

// Whether the capture rig makes this frame a rig frame (Docs/rig-frames.md): picked from the frame's game time like
// depth frames, about every other frame that is not a depth frame.
bool OrbitersRigFrame()
{
    return !OrbitersKitDepthFrame() && ((OrbitersHash(asuint(OrbitersTime())) >> 16) & 1u) != 0u;
}

// Eye depth (metres) of a top-down pixel from _CameraDepthTexture; +inf where nothing was drawn.
float OrbitersEyeDepthAt(uint2 p, uint width, uint height)
{
    float2 uv = float2((p.x + 0.5) / width, 1.0 - (p.y + 0.5) / height);
    float raw = SAMPLE_DEPTH_TEXTURE_LOD(_CameraDepthTexture, float4(uv, 0, 0));
    float z = LinearEyeDepth(raw);
    return z >= _ProjectionParams.z * 0.999 ? asfloat(0x7F800000u) : z;
}

// 16-bit depth code (Docs/depth-frames.md): 0 nothing there (beyond 1000 m: sky, far plane), 65535 unknown (nearer than
// 0.1 m: something is there, too near to place), 1..65534 inverse depth in between. Rounded as 1 + round(t * 65534), like
// the CPU twins: adding 1 before rounding rounded twice near powers of two.
uint OrbitersQuantizeDepth(float z)
{
    if (z > 1000.0) return 0u;
    if (!(z >= 0.1)) return 65535u;
    float t = (1.0 / z - 0.001) / (10.0 - 0.001);
    return (uint)clamp(1.0 + floor(t * 65534.0 + 0.5), 1.0, 65534.0);
}

float OrbitersLevelLinear(uint level)
{
    return GammaToLinearSpaceExact(0.2 * level);
}

// Depth layout: towards the edges, where chromatic aberration smears small cells, one tile of cells 2, 4 or 8 times as
// large takes the place of a block of 2 x 2, 4 x 4 or 8 x 8 grid tiles (region levels 1-3). Growth thresholds on
// OrbitersAberrationReach at a block's centre, and each level's pilot order (cell j of row 0 shows level
// ORBITERS_DEPTH_PILOTS[4 * level + j % 4]).
static const float ORBITERS_DEPTH_REACH[3] = { 1067.0, 1600.0, 2667.0 };
static const uint ORBITERS_DEPTH_PILOTS[16] = { 0u, 1u, 2u, 3u, 3u, 2u, 1u, 0u, 1u, 3u, 0u, 2u, 2u, 0u, 3u, 1u };
#define ORBITERS_NO_BLOCK 0xFFFFFFFFu

// PPv2's chromatic aberration segment at pixel (x, y) in pixels per unit of its amount (0.05 x intensity):
// |c|^2 |(c.x W, c.y H)| with c = 2 uv - 1, in the CPU twins' order.
float OrbitersAberrationReach(float x, float y, uint width, uint height)
{
    precise float w = (float)width;
    precise float h = (float)height;
    precise float cx = 2.0 * x / w - 1.0;
    precise float cy = 2.0 * y / h - 1.0;
    precise float px = cx * w;
    precise float py = cy * h;
    precise float reach = (cx * cx + cy * cy) * sqrt(px * px + py * py);
    return reach;
}

// First grid tile of the block of n tiles holding tile t, along an axis of `count` tiles (an even count), or
// ORBITERS_NO_BLOCK: blocks align from the nearer edge, and one crosses the middle where each half leaves n / 2 tiles.
uint OrbitersDepthBlockStart(uint t, uint count, uint n)
{
    uint mid = count / 2u;
    if (mid % n == n / 2u && t + n / 2u >= mid && t < mid + n / 2u) return mid - n / 2u;
    if (t < mid)
    {
        uint b = t / n * n;
        return b + n <= mid ? b : ORBITERS_NO_BLOCK;
    }
    if (t >= count - mid)
    {
        uint b = (count - 1u - t) / n * n;
        return b + n <= mid ? count - b - n : ORBITERS_NO_BLOCK;
    }
    return ORBITERS_NO_BLOCK;
}

// Whether the block of 2^level tiles from grid tile b becomes one tile of larger cells: inside the grid, clear of the
// strip's reserved rectangle, and with the aberration's reach at its centre at least the level's.
bool OrbitersDepthBlockGrows(uint2 b, uint level, uint tilePx, uint2 tiles, uint4 reserved, uint width, uint height)
{
    uint n = 1u << level;
    if (b.x + n > tiles.x || b.y + n > tiles.y) return false;
    uint4 r = uint4(b * tilePx, (b + n) * tilePx);
    if (r.x < reserved.z && reserved.x < r.z && r.y < reserved.w && reserved.y < r.w) return false;
    uint mid = n * tilePx / 2u;
    return OrbitersAberrationReach((float)(r.x + mid), (float)(r.y + mid), width, height) >= ORBITERS_DEPTH_REACH[level - 1u];
}

// Depth symbol colour of pixel p (top-down) in a depth frame. Call after OrbitersStripPixel returned false.
float3 OrbitersDepthPixel(uint2 p, uint width, uint height)
{
    OrbitersStripLayout l = OrbitersComputeStripLayout(width, height);
    uint cell = max(3u, height / 720u);
    uint tilePx = 16u * cell;
    uint2 tiles = uint2(width / tilePx, height / tilePx) & ~1u;
    uint2 t = p / tilePx;
    if (!l.valid || t.x >= tiles.x || t.y >= tiles.y) return 0;
    // The layout's tile holding p: the largest growing block, else the grid tile itself.
    uint level = 0u;
    uint2 origin = t;
    [unroll]
    for (uint k = 3u; k >= 1u; k--)
    {
        uint n = 1u << k;
        uint2 b = uint2(OrbitersDepthBlockStart(t.x, tiles.x, n), OrbitersDepthBlockStart(t.y, tiles.y, n));
        if (level == 0u && b.x != ORBITERS_NO_BLOCK && b.y != ORBITERS_NO_BLOCK && OrbitersDepthBlockGrows(b, k, tilePx, tiles, l.reserved, width, height))
        {
            level = k;
            origin = b;
        }
    }
    uint n = 1u << level;
    uint4 r = uint4(origin * tilePx, (origin + n) * tilePx);
    if (level == 0u && r.x < l.reserved.z && l.reserved.x < r.z && r.y < l.reserved.w && l.reserved.y < r.w) return 0;
    uint c = cell << level;
    uint col = (p.x - r.x) / c;
    uint row = (p.y - r.y) / c;
    uint symbol;
    if (row == 0u)
    {
        symbol = ORBITERS_DEPTH_PILOTS[4u * level + col % 4u];
    }
    else
    {
        uint e = (row - 1u) * 16u + col;
        uint s = e / 8u;
        // Sample s sits on every n-th cell of the block's sample grid, centred; that cell's pixel in its grid tile.
        uint2 g = origin * uint2(6u, 5u) + n * uint2(s % 6u, s / 6u) + n / 2u;
        uint2 sp = g / uint2(6u, 5u) * tilePx + (2u * (g % uint2(6u, 5u)) + 1u) * tilePx / uint2(12u, 10u);
        uint q = OrbitersQuantizeDepth(OrbitersEyeDepthAt(sp, width, height));
        symbol = (q >> (14u - 2u * (e % 8u))) & 3u;
    }
    return OrbitersLevelLinear(symbol);
}

// Clip position covering the whole screen for a quad mesh whose vertices span -0.5..0.5 (Unity's Quad).
float4 OrbitersFullscreenQuadVertex(float4 vertex)
{
    return float4(vertex.x * 2.0, vertex.y * 2.0, 0.5, 1.0);
}
