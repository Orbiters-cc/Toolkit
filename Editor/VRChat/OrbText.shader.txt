// The panel N4N0 projects (Orbiters contracts, Docs/orb.md): 4 rows of 32 cells, each holding a glyph of the panel's atlas
// in a material property the animator sets (_Char0 to _Char127, KillFrenzy Avatar Text's way of reaching them as one
// array). Changed for N4N0: a 16 × 16 atlas of tiles twice as tall as wide (tile g at column g mod 16, row g div 16 from the
// top), glyphs past 127 sent as negative values and wrapped (g − 256), a dark plate behind the text that is only as large
// as the text, the beam N4N0 projects it with (the second material, _Part 1), and the panel growing for people further
// away about its corner nearest its owner's face, so it never grows over that face. The panel and its beam are one skinned
// mesh: the plate and the beam's far end on the panel, the beam's near end on N4N0. Skinning puts the panel's scale in
// the vertices, so sizes here are in the panel's own metres and positions are only rescaled. Text seen from behind or in a
// mirror reads the right way round. Viewers who hide custom shaders see nothing of it (VRCFallback).
// OrbInstaller copies this file into each N4N0's folder as "N4N0 text.shader".

            float3 ViewerHead()
            {
                #if defined(USING_STEREO_MATRICES)
                return (unity_StereoWorldSpaceCameraPos[0] + unity_StereoWorldSpaceCameraPos[1]) * .5;
                #else

            float Glyph(float cell)
            {
                float g = round(_Chars[(int)cell]);
                return g < 0 ? g + 256 : g;
            }

            // The columns of the longest row and the rows down to the last one holding anything but spaces.
            float2 TextSize()
            {
                float columns = 0, rows = 0;
                [loop] for (int r = 0; r < 4; r++)
                {
                    float used = 0;
                    [loop] for (int c = 0; c < 32; c++) if (Glyph(r * 32 + c) != 0) used = c + 1;
                    columns = max(columns, used);
                    if (used > 0) rows = r + 1;
                }
                return float2(columns, rows);
            }

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_OUTPUT(v2f, o);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                float2 text = TextSize();
                float2 page = float2(32, 4) * _Cell.xy + 2 * _Pad;
                float2 plate = text * _Cell.xy + 2 * _Pad;
                bool beam = _Part > .5;
                float3 position = v.vertex.xyz;
                if (!beam || v.beam.y > .5)
                {
                    // The mesh holds a full page: this vertex moves to its corner of a plate as large as the text. Where it
                    // is in the mesh against where it is on a page gives the panel's scale (skinning puts it in the vertices).
                    float margin = beam ? 0 : Margin;
                    float2 onPage = float2(v.corner.x * (page.x + 2 * margin) - margin, (v.corner.y - 1) * (page.y + 2 * margin) + margin);
                    float scale = dot(onPage, onPage) > 1e-8 ? length(position.xy) / length(onPage) : 1;
                    float2 onPlate = float2(v.corner.x * (plate.x + 2 * margin) - margin, (v.corner.y - 1) * (plate.y + 2 * margin) + margin);
                    // People further away see it larger, by up to 1 + _Grow from three metres (in the panel's metres):
                    // its owner, about half a metre away, and people within arm's length see it as it is.
                    float away = length(ViewerHead() - unity_ObjectToWorld._m03_m13_m23) / max(scale, 1e-4);
                    float grow = 1 + _Grow * smoothstep(.8, 3, away);
                    position.xy = onPlate * scale * grow;
                    o.plate = float2(onPlate.x, -onPlate.y);
                }
                // Nothing to show: no plate, no beam.
                if (text.x == 0 || (beam && _Beam < .01)) position = 0;
                o.size = float4(text, plate);
                o.beam = v.beam;
                o.pos = UnityObjectToClipPos(float4(position, 1));
                return o;
            }

            fixed4 Plate(v2f i, bool flip)
            {
                float2 size = i.size.zw;
                // The plate: a rounded box with a thin bright edge and a soft glow past it.
                float2 q = abs(i.plate - size * .5) - (size * .5 - _Corner);
                float edge = length(max(q, 0)) + min(max(q.x, q.y), 0) - _Corner;
                float aa = max(fwidth(edge), 1e-5);
                float inside = 1 - smoothstep(-aa, aa, edge);
                float rule = 1 - smoothstep(.0005 - aa, .0005 + aa, abs(edge + .0011));
                float glow = (1 - inside) * exp(-max(edge, 0) / .0016) * .35;
                // Brighter towards its top, and the faint scan lines of a projection.
                float shade = 1.25 - .4 * saturate(i.plate.y / max(size.y, 1e-4));
                float scan = .93 + .07 * sin(i.plate.y * 2800 - _Time.y * 5);
                fixed4 colour = fixed4(_PlateColor.rgb * shade * scan, _PlateColor.a * inside);
                colour = lerp(colour, fixed4(_EdgeColor.rgb * 1.4, 1), rule * inside * .85);
                colour = lerp(colour, fixed4(_EdgeColor.rgb, 1), glow);
                colour.a = max(colour.a, glow);

                // The text: its cell, then the glyph's tile. Read from behind (or in a mirror), the text runs the other way.
                float2 grid = i.plate - _Pad;
                if (flip) grid.x = i.size.x * _Cell.x - grid.x;
                float2 cell = grid / _Cell.xy;
                float shown = all(cell >= 0) && cell.x < i.size.x && cell.y < i.size.y ? 1 : 0;
                float2 index = clamp(floor(cell), 0, float2(31, 3));
                float g = Glyph(index.y * 32 + index.x) * shown;
                float2 within = frac(cell);
                float2 uv = float2((fmod(g, 16) + within.x) / 16, 1 - (floor(g / 16) + within.y) / 16);
                // Gradients of the continuous position: no seams where cells meet.
                float2 dx = ddx(cell) * float2(1, -1) / 16, dy = ddy(cell) * float2(1, -1) / 16;
                fixed4 tile = tex2Dgrad(_MainTex, uv, dx, dy);
                float fill = saturate((tile.r - .5) / max(fwidth(tile.r), 1e-3) + .5);
                float rim = saturate((tile.a - .5) / max(fwidth(tile.a), 1e-3) + .5) * inside;
                fixed3 ink = lerp(_OutlineColor.rgb, _TextColor.rgb, fill);
                colour.rgb = lerp(colour.rgb, ink, rim);
                colour.a = max(colour.a, rim);
                return colour;
            }

            fixed4 Beam(v2f i)
            {
                float along = i.beam.y, across = i.beam.x;
                // A faint veil with fine bright lines along the pyramid's edges, a gentle shimmer running from N4N0 to the
                // panel; it leaves N4N0 from inside and thins out towards the panel.
                float edges = pow(1 - saturate(min(across, 1 - across) * 2), 28);
                float run = .8 + .2 * sin(along * 14 - _Time.y * 4);
                float fade = smoothstep(.02, .15, along) * (1 - .6 * along);
                return fixed4(_EdgeColor.rgb * (.045 + edges * .5) * run * fade * _Beam, 1);
            }

            fixed4 frag(v2f i, bool front : SV_IsFrontFace) : SV_Target
            {
                if (_Part > .5) return Beam(i);
                // KillFrenzy Avatar Text's flips: the back of the panel, and mirrors (oblique projections).
                bool mirror = unity_CameraProjection[2][0] != 0 || unity_CameraProjection[2][1] != 0;
                fixed4 colour = Plate(i, front == mirror);
                // Never true: makes Unity send every cell property.
                [branch] if (colour.a + i.pos.w == 65536) return SumCharacters();
                return colour;
            }
            ENDCG
        }
